อ่าน packet UDP แล้วแยกค่าด้วย comma ส่งต่อให้เกมผ่าน lock
byte[] data = udp.Receive(ref ep);
string[] p = Encoding.ASCII
    .GetString(data).Split(',');
bool  det = p[0] == "1";   // เจอมือ
float x   = float.Parse(p[1], inv);
float o   = float.Parse(p[3], inv); // openness
lock (lockObj) { latestPos = ...; }
ตัดสินกำ / แบ (Gesture + Openness)
ใช้ 2 สัญญาณร่วมกัน และต้องแบค้าง 0.08 วิ ถึงจะปล่อย กันกะพริบ
bool closed = gesture == "Closed_Fist"
           || open < gestureFistOpenness; // 1.35
bool opened = gesture == "Open_Palm"
           || open > gestureOpenOpenness; // 1.6
if (closed) { fistNow = true; openTimer = 0; }
else if (IsFist) {
    openTimer = opened ? openTimer + dt : 0;
    fistNow = openTimer < releaseDelay;
}
FistDown = fistNow && !IsFist; // เฟรมที่เพิ่งกำ
PROCESSOR · GameManager / IconSpawner
ค่าครองชีพ + เวลา 1 เดือน (ทุกเฟรม)
ตัวคูณ k ไต่จาก 1.0 เป็น 1.4 เมื่อใกล้สิ้นเดือน
float k = Mathf.Lerp(1f, endDrainMultiplier,
          1f - timeLeft / monthLength);
Change(-energyDrain * k * dt,
       -moneyDrain * k * dt,
       -happinessDrain * k * dt);
timeLeft -= dt;
if (timeLeft <= 0) End(EndKind.Victory);

if (tracker.FistDown) grabTimer = grabWindow;
if (grabTimer > 0) {
    grabTimer -= dt;
    if (TryGrab()) grabTimer = 0; // 1 กำ = 1 ชิ้น
}
// GameManager.ApplyItem()
if (item.kind == Kind.Work && BurnedOut
    && money > 0) money *= burnoutPay; // x0.5
if (Money <= 0) End(EndKind.NoMoney);
else if (Energy <= 0) End(EndKind.NoHealth);
